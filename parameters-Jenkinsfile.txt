pipeline {
    agent none

    parameters {
        // Remove these two if you already define them elsewhere.
        string(name: 'REPO_URL', defaultValue: '', description: 'SVN repo URL recorded on the board')
        string(name: 'REVISION', defaultValue: '', description: 'SVN revision to build / record')
    }

    stages {
        stage('Build') {
            steps {
                script {
                    runCmdOnCheckout('make clean && make',
                        pre: 'gcc --version',
                        timeout: 100,
                        post: 'svn info || true',
                        cleanup: { cleanWs() })
                }
            }
        }
    }

    post {
        always {
            node('built-in') {                 // controller, so userContent is writable
                lock('buildboard') {           // serialize across all pipelines; drop if no Lockable Resources plugin
                    script {
                        // ---- buildboard builder: append one record ----
                        def uc   = currentBuild.getBuildCauses().find { it._class?.contains('UserIdCause') }
                        def user = (uc?.userId ?: uc?.userName ?: 'automated').replace('|','/')
                        def ts   = new Date().format("yyyy-MM-dd'T'HH:mm:ss'Z'", TimeZone.getTimeZone('UTC'))
                        def repo = (params.REPO_URL ?: env.SVN_URL_1   ?: 'unknown').replace('|','/')
                        def rev  = (params.REVISION ?: env.SVN_REVISION ?: 'unknown').toString()
                        def res  = currentBuild.currentResult           // SUCCESS / FAILURE / UNSTABLE
                        def line = [ts, repo, rev, user, res, env.JOB_NAME].join('|')

                        def home  = env.JENKINS_HOME
                        def board = "${home}\\userContent\\buildboard.csv"

                        withEnv(["BOARD_LINE=${line}", "BOARD_FILE=${board}"]) {
                            bat '''@echo off
setlocal enabledelayedexpansion
if not exist "%BOARD_FILE%" echo timestamp^|repo_url^|revision^|user^|result^|job>"%BOARD_FILE%"
echo !BOARD_LINE!>>"%BOARD_FILE%"'''
                        }

                        // ---- board render: regenerate HTML, non-fatal ----
                        def rc = bat(returnStatus: true, script:
                            "python \"${home}\\userContent\\render_buildboard.py\" " +
                            "-i \"${home}\\userContent\\buildboard.csv\" " +
                            "-o \"${home}\\userContent\\buildboard.html\" -t \"Build Board\"")
                        if (rc != 0) {
                            echo "buildboard render failed (rc=${rc}); record was still appended"
                        }
                    }
                }
            }
        }
    }
}
